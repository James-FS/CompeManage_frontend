import { useRouter } from 'vue-router'

  School, UploadFilled, ArrowLeft, 
  Plus, Delete, Trophy, Document
} from '@element-plus/icons-vue'

  leader: { name: '张三', stuId: '2023001', phone: '', email: '' },

// 2. 校验规则

  'leader.email': [{ required: true, message: '必填', trigger: 'blur' }],
}

// 3. 逻辑操作
const addMember = () => {
  if (formData.members.length >= 4) return ElMessage.warning('人数已达上限')
  formData.members.push({ name: '', stuId: '', phone: '' })
}
const removeMember = (index) => formData.members.splice(index, 1)

const submitForm = async (formEl) => {
  if (!formEl) return
  await formEl.validate((valid) => {
    if (valid) ElMessage.success('提交成功')
  })
}
